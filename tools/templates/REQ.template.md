```bash
python3 -m venv .tools/wdvenv && .tools/wdvenv/bin/pip install wavedrom   # 一次性准备
.tools/wdvenv/bin/wavedrompy -i 01-requirements/<REQ 文件名去扩展名>/scenario-1-xxx.json \
                             -s 01-requirements/<REQ 文件名去扩展名>/scenario-1-xxx.svg
```
